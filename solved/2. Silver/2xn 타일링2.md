#include <iostream>
#include <vector>
using namespace std;

int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);
	
	int t;
	cin >> t;
	
	vector<int> v = vector<int>(1000, 0);
	v[2] = 3;
	v[1] = 1;

	for (int i = 3; i <= t; i++)
	{
		v[i] += (v[i - 1] + v[i - 2] * 2) % 10007;
	}

	cout << v[t];
}
