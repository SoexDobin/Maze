```c++
#include <iostream>
#include <vector>
using namespace std;

int main() {
    // 입력 받기
    int cardCount, amount;
    cin >> cardCount >> amount;

    vector<int> cards(cardCount);
    for (int i = 0; i < cardCount; i++) {
        cin >> cards[i];
    }

    int best = 0;

    // 3중 루프를 사용하여 모든 가능한 조합을 탐색
    for (int i = 0; i < cardCount; i++) {
        for (int j = 0; j < cardCount; j++) {
            if (i == j) continue;                           // 중복 인덱스 비교X
            for (int k = 0; k < cardCount; k++) {
                if (j == k || i == k) continue;             // 중복 인덱스 비교X
                int sum = cards[i] + cards[j] + cards[k];
                if (sum <= amount && sum > best) {
                    best = sum;
                }
            }
        }
    }

    cout << best << endl;
    return 0;
}
```
