#include <iostream>
#include <vector>
using namespace std;

int Fac(const vector<vector<int>>& v, const int& find
	, const int& fromY, const int& toY, const int& fromX, const int& toX)
{
	int allSame = 1;
	for (int i = fromY; i <= toY; i++)
	{
		for (int j = fromX; j <= toX; j++)
		{
			if (v[i][j] != find)
				allSame = 0;
		}
		if (allSame == 0) break;
	}

	int divY = (toY + fromY) / 2;
	int divX = (toX + fromX) / 2;
	
	if (toY == fromY || toX == fromX) return allSame;

	if (allSame) return allSame;
	else
		return 
		Fac(v, find, fromY, divY, fromX, divX)
		+ Fac(v, find, fromY, divY, divX + 1, toX)
		+ Fac(v, find, divY + 1, toY, fromX, divX)
		+ Fac(v, find, divY + 1, toY, divX + 1, toX);
}

int main()
{
	cin.tie(nullptr);
	ios_base::sync_with_stdio(false);

	vector<vector<int>> v;
	int t;
	cin >> t;
	v.resize(t);
	for (int i = 0; i < t; i++)
	{
		for (int j = 0; j < t; j++)
		{
			int x;
			cin >> x;
			v[i].push_back(x);
		}
	}
	cout << Fac(v, 0, 0, t - 1, 0, t - 1) << "\n";
	cout << Fac(v, 1, 0, t - 1, 0, t - 1) << "\n";
}
