#include <iostream>
#include <vector>
#include <queue>
using namespace std;

void DFS(vector<vector<int>>& v, vector<bool>& visited, const int& idx)
{
	if (visited[idx]) return;
	visited[idx] = true;
	int size = static_cast<int>(v.size());

	for (int i = 1; i < size; i++)
	{
		if (!visited[i] && v[idx][i] == 1)
		{
			v[idx][i] = 0;
			v[i][idx] = 0;
			DFS(v, visited, i);
		}
	}
}


int main()
{
	cin.tie(nullptr);
	ios_base::sync_with_stdio(false);

	vector<vector<int>> v;
	vector<bool> visited;
	int node, edge;
	cin >> node >> edge;

	v = vector<vector<int>>(node + 1, vector<int>(node + 1, 0));
	visited = vector<bool>(node + 1);
	int start;
	for (int i = 0; i < edge; i++)
	{
		int x, y;
		cin >> x >> y;
		v[x][y] = 1;
		v[y][x] = 1;
	}

	int component = 0;
	for (int i = 1; i <= node; i++)
	{
		if (!visited[i])
		{
			DFS(v, visited, i);
			component++;
		}
	}
	
	cout << component;
}
