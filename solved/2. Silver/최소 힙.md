```c++
#include <iostream>
#include <queue>
using namespace std;

int main()
{
	cin.tie(nullptr);
	ios_base::sync_with_stdio(false);

	int t;
	cin >> t;
	priority_queue<int, vector<int>, greater<int>> q;
	for (int i = 0; i < t; i++)
	{
		int x;
		cin >> x;
		if (x > 0)
			q.push(x);
		else
		{
			if (q.size() == 0)
				cout << 0 << "\n";
			else
			{
				int n = q.top();
				cout << n << "\n";
				q.pop();
			}
		}
	}
}
```
