```c++
#include <iostream>
#include <vector>
using namespace std;

int main() {
   
    vector<vector<int>> v;
    int n;
    int index = 0;

    
    while (true)
    {
        cin >> n;
        if (n == 0) break;

        v.push_back(vector<int>());
        while (n != 0)
        {   // 10으로 계속해서 나눠 1자리 수로 등록 
            v[index].push_back(n % 10);
            n = n / 10;
        }
        index++;
    }
    
    
    vector<bool> pelVec = vector<bool>( v.size(), true);
    for (int i = 0; i < v.size(); i++)
    {
        int halfSize = v[i].size() / 2;
        int backIndex = v[i].size() - 1;

        bool pel = true;
        for (int j = 0; j < halfSize; j++)
        {
            if (v[i][j] != v[i][backIndex - j])
                pel = false;
        }
        pelVec[i] = pel;
    }

    for (int i = 0; i < pelVec.size(); i++)
    {
        if (pelVec[i])
            cout << "yes" << endl;
        else
            cout << "no" << endl;
    }
}
```
