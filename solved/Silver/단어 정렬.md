```c++
#include <iostream>
#include <string>
#include <string.h>
#include <vector>
using namespace std;

int main()
{
	vector<string> s;
	int size;
	
	cin >> size;
	s.resize(size);
	for (int i = 0; i < size; i++) 
	{
		cin >> s[i];
	}
	
	for (int i = 1; i < size; i++)
	{
		string cmpstr = s[i];

		int j;
		for (j = i - 1; j >= 0; j--)
		{				
			// 현재 단어가 더 짧으면
			if (s[j].size() > cmpstr.size())
				s[j + 1] = s[j];
			else if (s[j].size() == cmpstr.size())
			{
				int k = strcmp(&s[j].front(), &cmpstr.front());

				if (k > 0)
					s[j + 1] = s[j];
				else
					break;
			}
			else
				break;
		}
		s[j + 1] = cmpstr;
	}
	
	string prev = "";
	for (int i = 0; i < size; i++)
	{
		if (s[i] == prev) continue;
		cout << s[i] << endl;
		prev = s[i];
	}
}
```
### [1181] 단어정렬
1. ```strcmp()```함수 ++17 기준 <string.h> 사용해야 한다.
2. 정렬 알고리즘 이 코드의 핵심이다.
3. ```strcmp(const char* str1, const char* str2)```함수는 
    - 0 -> 두 문자열이 같음
    - 양수 str1이 str2보다 사전순 뒤에 있음
    - 음수 str1이 str2보다 사전순 앞에 있음
