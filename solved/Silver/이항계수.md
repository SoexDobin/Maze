```c++
#include <iostream>
using namespace std;

int Fac(int n, int k)
{
	int facN = 1;
	int facK = 1;
	int minFac = 1;

	int MinNK = n - k;
	for (int i = MinNK; i > 0; i--)
	{
		if (minFac == 1)
			minFac = i;
		else
			minFac *= i;
	}
	for (int i = n; i > 0; i--)
	{
		if (facN == 1)
			facN = i;
		else
			facN *= i;
	}
	for (int i = k; i > 0; i--)
	{
		if (facK == 1)
			facK = i;
		else
			facK *= i;
	}
	return facN / (facK * minFac);
}

int main()
{
	int n, k;
	cin >> n >> k;
	cout << Fac(n, k) << endl;
}
```
