```c++
#include <iostream>
#include <cmath>
using namespace std;

int main()
{
	unsigned int adj, opp, hyp;

	while (true)
	{
		cin >> adj >> opp >> hyp;

		if (hyp < adj)
			::swap(hyp, adj);
		if (hyp < opp)
			::swap(hyp, opp);

		int amount = static_cast<int>(pow(adj, 2) + pow(opp, 2));

		if (adj + opp + hyp == 0)
			break;
		else if (sqrt(amount) == hyp)
			cout << "right" << endl;
		else
			cout << "wrong" << endl;
	}
}
```
